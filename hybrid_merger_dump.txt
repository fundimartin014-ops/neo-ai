 165:     async def _call_model(self, model_config: Dict, prompt: str, system: Optional[str] = None) -> ModelOutput:
 166:         """Call a single cloud model via OpenAI-compatible API."""
 167:         import aiohttp
 168: 
 169:         provider = model_config.get("provider", "unknown")
 170:         model_id = model_config["id"]
 171:         base_url = model_config.get("base_url", "")
 172:         env_key = model_config.get("env_key") or model_config.get("env")
 173: 
 174:         api_key = self._get_api_key(env_key) if env_key else None
 175:         if env_key and not api_key:
 176:             return ModelOutput(
 177:                 model_id=model_id, provider=provider,
 178:                 output="", latency=0, error=f"Missing API key: {env_key}"
 179:             )
 180: 
 181:         # Handle Cloudflare special URL
 182:         if "{account_id}" in base_url:
 183:             account_id = os.getenv("CF_ACCOUNT_ID", "")
 184:             base_url = base_url.replace("{account_id}", account_id)
 185: 
 186:         # Fallback: infer base_url from provider if missing
 187:         if not base_url:
 188:             _provider_urls = {
 189:                 "groq": "https://api.groq.com/openai/v1",
 190:                 "openrouter": "https://openrouter.ai/api/v1",
 191:                 "deepinfra": "https://api.deepinfra.com/v1/openai",
 192:                 "mistral": "https://api.mistral.ai/v1",
 193:                 "fireworks": "https://api.fireworks.ai/inference/v1",
 194:                 "novita": "https://api.novita.ai/v3/openai",
 195:                 "hyperbolic": "https://api.hyperbolic.xyz/v1",
 196:                 "siliconflow": "https://api.siliconflow.cn/v1",
 197:                 "friendli": "https://api.friendli.ai/serverless/v1",
 198:             }
 199:             base_url = _provider_urls.get(provider, "")
 200:         url = f"{base_url}/chat/completions"
 201:         payload = self._build_openai_payload(prompt, model_id, system)
 202:         headers = {"Content-Type": "application/json"}
 203:         if api_key:
 204:             headers["Authorization"] = f"Bearer {api_key}"
 205: 
 206:         start = time.time()
 207:         try:
 208:             timeout = aiohttp.ClientTimeout(total=self.timeout)
 209:             async with aiohttp.ClientSession(timeout=timeout) as session:
 210:                 async with session.post(url, json=payload, headers=headers) as resp:
 211:                     if resp.status != 200:
 212:                         text = await resp.text()
 213:                         return ModelOutput(
 214:                             model_id=model_id, provider=provider,
 215:                             output="", latency=time.time()-start,
 216:                             error=f"HTTP {resp.status}: {text[:200]}"
 217:                         )
 218:                     data = await resp.json()
 219:                     content = data["choices"][0]["message"]["content"] if "choices" in data else str(data)
 220:                     tokens = data.get("usage", {}).get("total_tokens", 0)
 221:                     return ModelOutput(
 222:                         model_id=model_id, provider=provider,
 223:                         output=content.strip(), latency=time.time()-start,
 224:                         tokens_used=tokens, raw_response=data
 225:                     )
 226:         except asyncio.TimeoutError:
 227:             return ModelOutput(
 228:                 model_id=model_id, provider=provider,
 229:                 output="", latency=time.time()-start, error="Timeout"
 230:             )
 231:         except Exception as e:
 232:             return ModelOutput(

 405:     async def _apply_strategy(self, outputs: List[ModelOutput], strategy: str, original_prompt: str) -> str:
 406:         """Apply the chosen merge strategy."""
 407: 
 408:         # Filter out errors
 409:         valid = [o for o in outputs if not o.error and o.output]
 410:         if not valid:
 411:             return "[HybridMerger] All models failed."
 412: 
 413:         if strategy == "cascade":
 414:             # Chain: pass output of model i to model i+1 as a refinement prompt
 415:             current = valid[0].output
 416:             for i, out in enumerate(valid[1:], 1):
 417:                 refine_prompt = f"Improve and refine the following response to the query: {original_prompt}\n\nCURRENT RESPONSE:\n{current}\n\nREFINED RESPONSE:"
 418:                 # We'd need to re-call the model here; for now, use merger
 419:                 merged = await self._call_merger(
 420:                     refine_prompt,
 421:                     "You are a refinement engine. Improve clarity, accuracy, and depth."
 422:                 )
 423:                 if merged:
 424:                     current = merged
 425:             return current
 426: 
 427:         # Build strategy-specific prompt
 428:         builders = {
 429:             "synthesis": self._build_synthesis_prompt,
 430:             "consensus": self._build_consensus_prompt,
 431:             "debate": self._build_debate_prompt,
 432:             "ensemble": self._build_ensemble_prompt,
 433:             "best_of_all": self._build_best_of_all_prompt,
 434:             "parallel": self._build_parallel_prompt,
 435:         }
 436: 
 437:         builder = builders.get(strategy, self._build_synthesis_prompt)
 438:         meta_prompt = builder(valid, original_prompt)
 439: 
 440:         system = "You are the NEO Hybrid Synthesis Engine. You merge multiple AI outputs into one superior response."
 441:         merged = await self._call_merger(meta_prompt, system)
 442: 
 443:         if merged:
 444:             return merged
 445: 
 446:         # Fallback: simple concatenation with headers
 447:         fallback = f"# Hybrid Output ({strategy})\n\n"
 448:         for o in valid:
 449:             fallback += f"## {o.provider}/{o.model_id}\n{o.output}\n\n"
 450:         return fallback

 452:     async def hybrid_generate(
 453:         self,
 454:         prompt: str,
 455:         model_pool: Optional[List[str]] = None,
 456:         strategy: str = "synthesis",
 457:         auto_pool: Optional[str] = None,
 458:         pool_size: int = 5,
 459:         system: Optional[str] = None,
 460:         use_cache: bool = True,
 461:     ) -> HybridResult:
 462:         """
 463:         Generate a hybrid response by running multiple models and merging outputs.
 464: 
 465:         Args:
 466:             prompt: The user query/prompt.
 467:             model_pool: List of model IDs to use. If None, auto-selects based on auto_pool.
 468:             strategy: Merge strategy — synthesis, consensus, debate, ensemble, best_of_all, parallel, cascade.
 469:             auto_pool: Auto-select pool by type: "chat", "coding", "reasoning", "vision", "image_gen", etc.
 470:             pool_size: Number of models to run in parallel (default 5).
 471:             system: Optional system prompt for all models.
 472:             use_cache: Whether to cache results.
 473: 
 474:         Returns:
 475:             HybridResult with merged_output and individual outputs.
 476:         """
 477: 
 478:         # Cache key
 479:         cache_key = hashlib.sha256(f"{prompt}:{strategy}:{','.join(model_pool or [])}:{auto_pool}:{pool_size}".encode()).hexdigest()
 480:         if use_cache and cache_key in self._cache:
 481:             logger.info("[HybridMerger] Cache hit")
 482:             return self._cache[cache_key]
 483: 
 484:         # Resolve model pool
 485:         resolved_models: List[Dict] = []
 486: 
 487:         if model_pool:
 488:             for hint in model_pool:
 489:                 m = self._resolve_model(hint)
 490:                 if m:
 491:                     resolved_models.append(m)
 492:         elif auto_pool and self.catalog:
 493:             pool_map = {
 494:                 "chat": "chat", "coding": "coding", "reasoning": "reasoning",
 495:                 "vision": "vision", "image_gen": "image-gen", "video_gen": "video-gen",
 496:                 "embedding": "embedding", "image_edit": "image-edit", "image_describe": "image-describe",
 497:             }
 498:             mtype = pool_map.get(auto_pool, auto_pool)
 499:             candidates = self.catalog["by_type"](mtype)
 500:             # Prioritize free tiers, then pick diverse families
 501:             free_models = [m for m in candidates if m.get("tier") in ("free", "free_fast", "free_no_key")]
 502:             credit_models = [m for m in candidates if m.get("tier") == "free_credit"]
 503:             tier_models = [m for m in candidates if m.get("tier") == "free_tier"]
 504: 
 505:             selected = []
 506:             seen_families = set()
 507:             for source in (free_models, tier_models, credit_models):
 508:                 for m in source:
 509:                     fam = m.get("family", "unknown")
 510:                     if fam not in seen_families and len(selected) < pool_size:
 511:                         selected.append(m)
 512:                         seen_families.add(fam)
 513:                 if len(selected) >= pool_size:
 514:                     break
 515:             resolved_models = selected[:pool_size]
 516: 
 517:         # ── Filter out decommissioned Groq models ──
 518:         # If the catalog still lists retired Groq IDs, swap them for the
 519:         # live default. Any non-Groq models pass through untouched.
 520:         _filtered = []
 521:         for _m in resolved_models:
 522:             if _m.get("provider") == "groq":
 523:                 _mid = _m.get("id", "")
 524:                 if nmh.is_dead("groq", _mid):
 525:                     logger.info(
 526:                         f"[HybridMerger] Dropping dead Groq model '{_mid}'"
 527:                     )
 528:                     try:
 529:                         nmh.record_dead("groq", _mid)
 530:                     except Exception:
 531:                         pass
 532:                     continue
 533:             _filtered.append(_m)
 534:         resolved_models = _filtered
 535: 
 536:         # ── Diversity check ──
 537:         # If the catalog collapsed to fewer than 2 unique (provider, id)
 538:         # pairs (which happens when it's stale and every Groq entry hits
 539:         # the _GROQ_FALLBACK swap), replace with the curated pool so the
 540:         # hybrid actually runs across multiple live models from multiple
 541:         # providers. Only applies when caller used auto_pool — explicit
 542:         # model_pool choices are respected untouched.
 543:         if not model_pool:
 544:             _unique = {(mm.get("provider"), mm.get("id")) for mm in resolved_models}
 545:             if len(_unique) < 2:
 546:                 logger.warning(
 547:                     f"[HybridMerger] Catalog gave {len(_unique)} unique model(s), "
 548:                     f"swapping to curated pool for real diversity"
 549:                 )
 550:                 resolved_models = list(self._CURATED_POOL)[:max(pool_size, 2)]
 551:                 _unique = {(mm.get("provider"), mm.get("id")) for mm in resolved_models}
 552:                 logger.info(
 553:                     f"[HybridMerger] Curated pool selected: {sorted(_unique)}"
 554:                 )
 555:         # ── end diversity check ──
 556:         # ── neo_patch_v9_1: pad to pool_size from known-good pool ──
 557:         if not model_pool and len(resolved_models) < pool_size:
 558:             _have = {(m.get("provider"), m.get("id")) for m in resolved_models}
 559:             try:
 560:                 _pad = nmh.get_pool(pool_size)
 561:             except Exception:
 562:                 _pad = []
 563:             for _cand in _pad:
 564:                 _ck = (_cand.get("provider"), _cand.get("id"))
 565:                 if _ck in _have:
 566:                     continue
 567:                 resolved_models.append(_cand)
 568:                 _have.add(_ck)
 569:                 if len(resolved_models) >= pool_size:
 570:                     break
 571:         # ── end padding ──
 572: 
 573:         # ── end filter ──
 574: 
 575:         if not resolved_models:
 576:             return HybridResult(
 577:                 strategy=strategy, prompt=prompt,
 578:                 outputs=[], merged_output="[HybridMerger] No models resolved.",
 579:                 merge_latency=0, metadata={"error": "No models available"}
 580:             )
 581: 
 582:         logger.info(f"[HybridMerger] Running {len(resolved_models)} models with strategy={strategy}")
 583: 
 584:         # Run all models in parallel
 585:         start = time.time()
 586:         tasks = [self._call_model(m, prompt, system) for m in resolved_models]
 587:         outputs = await asyncio.gather(*tasks, return_exceptions=True)
 588: 
 589:         # ── neo_patch_v9_1: record outcomes for pool self-heal ──
 590:         for _m, _out in zip(resolved_models, outputs):
 591:             _prov = _m.get("provider", "?")
 592:             _mid  = _m.get("id", "?")
 593:             try:
 594:                 if isinstance(_out, Exception):
 595:                     _err = str(_out)
 596:                     if "429" in _err or "rate" in _err.lower():
 597:                         nmh.record_throttled(_prov, _mid)
 598:                     elif ("404" in _err or "not_found" in _err
 599:                           or "does not exist" in _err.lower()):
 600:                         nmh.record_dead(_prov, _mid)
 601:                 else:
 602:                     nmh.record_success(_prov, _mid)
 603:             except Exception:
 604:                 pass
 605:         # ── end outcome recording ──
 606: 
 607:         # Normalize outputs
 608:         normalized = []
 609:         for i, result in enumerate(outputs):
 610:             if isinstance(result, Exception):
 611:                 normalized.append(ModelOutput(
 612:                     model_id=resolved_models[i]["id"],
 613:                     provider=resolved_models[i].get("provider", "unknown"),
 614:                     output="", latency=0, error=str(result)
 615:                 ))
 616:             else:
 617:                 normalized.append(result)
 618: 
 619:         # Apply merge strategy
 620:         merge_start = time.time()
 621:         merged = await self._apply_strategy(normalized, strategy, prompt)
 622:         merge_latency = time.time() - merge_start
 623: 
 624:         result = HybridResult(
 625:             strategy=strategy,
 626:             prompt=prompt,
 627:             outputs=normalized,
 628:             merged_output=merged,
 629:             merge_latency=merge_latency,
 630:             metadata={
 631:                 "total_models": len(resolved_models),
 632:                 "successful": len([o for o in normalized if not o.error]),
 633:                 "failed": len([o for o in normalized if o.error]),
 634:                 "total_latency": time.time() - start,
 635:                 "cache_key": cache_key,
 636:                 "errors": {
 637:                     f"{o.provider}/{o.model_id}": o.error
 638:                     for o in normalized if o.error
 639:                 },
 640:                 "successes": {
 641:                     f"{o.provider}/{o.model_id}": f"{len(o.output)} chars, {o.latency:.2f}s"
 642:                     for o in normalized if not o.error
 643:                 },
 644:             }
 645:         )
 646: 
 647:         if use_cache:
 648:             self._cache[cache_key] = result
 649: 
 650:         return result
